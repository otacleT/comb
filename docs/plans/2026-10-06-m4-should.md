```ts
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { GENERATED_MARKER, isUnmodifiedGenerated, type AgentAdapter } from '../../agents/adapter.js';
import { normalizeText } from '../markdown/text.js';
import { defaultConfig, renderConfig } from './config.js';

export interface InitResult {
  created: string[];
  updated: string[];
  skipped: string[];
  /** Comb-generated files the user edited; kept as they are. */
  modified: string[];
}

type Put = (relPath: string, content: string, generated: boolean) => void;

const emptyResult = (): InitResult => ({ created: [], updated: [], skipped: [], modified: [] });

/** Writes a file unless that would overwrite a user file or a generated file the user edited. */
function writer(root: string, result: InitResult): Put {
  return (relPath, content, generated) => {
    const p = join(root, relPath);
    if (!existsSync(p)) {
      mkdirSync(dirname(p), { recursive: true });
      writeFileSync(p, content);
      result.created.push(relPath);
      return;
    }
    const existing = readFileSync(p, 'utf8');
    if (normalizeText(existing) === content) {
      result.skipped.push(relPath);
      return;
    }
    if (generated && isUnmodifiedGenerated(existing)) {
      writeFileSync(p, content);
      result.updated.push(relPath);
      return;
    }
    if (generated && existing.includes(GENERATED_MARKER)) {
      result.modified.push(relPath);
      return;
    }
    result.skipped.push(relPath);
  };
}

interface AgentOptions {
  version: string;
  templatesDir: string;
  agents: AgentAdapter[];
}

const putAgentFiles = (put: Put, opts: AgentOptions): void => {
  for (const agent of opts.agents) {
    for (const f of agent.files(opts.templatesDir, opts.version)) put(f.path, f.content, true);
  }
};

export function initProject(root: string, opts: AgentOptions & { language: string }): InitResult {
  const result = emptyResult();
  const put = writer(root, result);
  const config = defaultConfig(opts.language);
  config.agents = opts.agents.map((a) => a.id);
  put('comb/config.yaml', renderConfig(config), false);
  put('comb/decisions.md', '# Decisions\n', false);
  put('comb/specs/.gitkeep', '', false);
  put('comb/changes/archive/.gitkeep', '', false);
  putAgentFiles(put, opts);
  return result;
}

/** S3 (`comb update`): regenerates the agent files only. The config, the specs and files the user edited stay as they are. */
export function updateAgentFiles(root: string, opts: AgentOptions): InitResult {
  const result = emptyResult();
  putAgentFiles(writer(root, result), opts);
  return result;
}
```

```ts
import type { Command } from 'commander';
import { AGENTS } from '../../agents/registry.js';
import { hasErrors } from '../../core/diagnostics.js';
import { loadProject } from '../../core/project/load.js';
import { updateAgentFiles } from '../../core/project/init.js';
import { TEMPLATES_DIR } from '../../templates-dir.js';
import { VERSION } from '../../version.js';
import type { CliContext } from '../app.js';
import { envelope } from '../output.js';
import { describeInit } from './init.js';
import { finish, requireRoot, usageError } from './shared.js';

export function registerUpdate(program: Command, ctx: CliContext): void {
  program
    .command('update')
    .description('regenerate the agent skills and commands for the installed comb version')
    .option('--json', 'JSON output')
    .action((opts: { json?: boolean }) => {
      const json = opts.json === true;
      const root = requireRoot(ctx, 'update', json);
      if (!root) return;
      const project = loadProject(root);
      if (hasErrors(project.diagnostics)) {
        finish(ctx, envelope('update', null, project.diagnostics), json, () => '');
        return;
      }
      const unknown = project.config.agents.find((id) => AGENTS[id] === undefined);
      if (unknown !== undefined) {
        usageError(ctx, 'update', json, 'UNKNOWN_AGENT', `unknown agent "${unknown}" in comb/config.yaml`, `Use one of: ${Object.keys(AGENTS).join(', ')}`);
        return;
      }
      const agents = project.config.agents.map((id) => AGENTS[id]!);
      finish(ctx, envelope('update', updateAgentFiles(root, { version: VERSION, templatesDir: TEMPLATES_DIR, agents }), []), json, describeInit);
    });
}
```

# Comb

From tangled ideas to decided specs.

Comb is a spec-driven development CLI for AI coding agents. It keeps human-readable specs in your repository and drives a convergence loop: review → answerable questions → recorded decisions → updated specs. The CLI does the deterministic work (parsing, validation, ids, gates, merging); the agent writes and reviews through Claude Code skills.

## Install

```bash
npm install -g @otaclet/comb    # Node 22.12 or later
cd /path/to/your/project
comb init --language en          # or ja
```

`comb init` creates `comb/` (config, specs, decisions, changes) and installs the Claude Code skills and `/comb:*` commands. After upgrading Comb, run `comb update` to refresh them; files you edited are kept.

## The loop

| Step | In Claude Code | What happens |
|---|---|---|
| Explore | `/comb:explore` | think the idea through; no files are written |
| Propose | `/comb:propose <idea>` | a change with a proposal, a spec delta with stable ids (`REQ-<PREFIX>-<NNN>`) and tasks |
| Review | `/comb:review` | four lenses (consistency, formalization, edge cases, change impact) → `review.md` and answerable questions with options and a recommendation |
| Clarify | `/comb:clarify` | your answers become decisions (`D-1`…) and delta edits; the review goes stale until it is redone |
| Apply | `/comb:apply` | implement the tasks |
| Archive | `/comb:archive` | merge the delta into `comb/specs/`, give the decisions permanent ids (`D-001`…) in `comb/decisions.md`, and move the change to `comb/changes/archive/` |

`comb status` always tells you, and the agent, the next step. Open high-severity questions block apply until you answer or defer them; archive needs every question answered or deferred.

## Commands

| Command | What it does |
|---|---|
| `comb init [--language <lang>]` | create `comb/` and install the agent skills |
| `comb update` | regenerate the skills for the installed version |
| `comb new change <id>` | create a change folder |
| `comb status [--change <id>]` | artifact states, open questions and the next step |
| `comb instructions <artifact> --change <id>` | instructions for the agent (proposal, delta, tasks, review, clarify, apply) |
| `comb validate [--change <id>]` | check specs, changes, questions and decisions |
| `comb mark-reviewed --change <id>` | record that the review matches the current delta |
| `comb archive <id> [--yes] [--allow-dropped-scenarios] [--allow-base-drift]` | merge a finished change |
| `comb list` / `comb show <REQ-…\|D-NNN\|change-id>` | list and look up specs, decisions and changes |

Every command accepts `--json` and returns `{ "schema": "comb.v1", "command", "ok", "data", "diagnostics" }`. Exit codes: 0 ok, 1 a check failed or a gate is not met, 2 usage error, 3 internal error.

// Pure helpers for bench/run.mjs, kept apart so they can be unit-tested.

/** Turns, cost, minutes, tool errors and the final text of a `claude -p --output-format stream-json` transcript. */
export function resultOf(jsonl) {
  let toolErrors = 0;
  let result = null;
  for (const line of jsonl.split('\n')) {
    if (!line.trim()) continue;
    const e = JSON.parse(line);
    if (e.type === 'user' && Array.isArray(e.message?.content)) {
      toolErrors += e.message.content.filter((c) => c.type === 'tool_result' && c.is_error).length;
    }
    if (e.type === 'result') result = e;
  }
  if (!result) return null;
  return {
    turns: result.num_turns,
    costUsd: result.total_cost_usd,
    minutes: Math.round(result.duration_ms / 6000) / 10,
    toolErrors,
    text: result.result ?? '',
  };
}

/** The ids in the first column of an expected-highs.md table (`| BL-01 | … |`). */
export const answerKeys = (markdown) => [...markdown.matchAll(/^\|\s*([A-Z]+-\d+)\s*\|/gm)].map((m) => m[1]);

/** The strict matching rule used since F2 (one question per key, same root cause, failure made visible). */
export function graderPrompt({ spec, answerKey, questions }) {
  return [
    'You grade the recall of one AI spec review against a fixed answer key. Be strict and do not use any tools.',
    '',
    'A key item is matched when one QUESTION (any severity) raises the same root cause: the same missing or wrong rule leading to the same kind of bad outcome, and its sequence or text makes that failure visible. A question about a neighbouring issue that does not state the failure is not a match. One question may match several keys if it explicitly covers each.',
    '',
    'Reply with only a JSON object inside a ```json fence: {"matches":[{"key":"<id>","question":"Q-<n>" or null,"quote":"at most 25 words from the question that prove the match, or empty"}]}, one entry per key, in key order.',
    '',
    '## Spec',
    spec,
    '',
    '## Answer key',
    answerKey,
    '',
    '## Questions to grade',
    questions,
  ].join('\n');
}

/** The grader's verdict for every key; keys the grader skipped count as not matched. */
export function parseGrade(text, keys) {
  const m = /```json\s*([\s\S]*?)```/.exec(text) ?? /(\{[\s\S]*\})/.exec(text);
  if (!m) throw new Error('grader returned no JSON');
  const { matches } = JSON.parse(m[1]);
  const byKey = new Map((matches ?? []).map((x) => [x.key, x]));
  return keys.map((key) => ({ key, question: byKey.get(key)?.question ?? null, quote: byKey.get(key)?.quote ?? '' }));
}

export function summarize(subject, runs, keys) {
  const recalls = runs.map((r) => r.matches.filter((x) => x.question).length);
  return {
    subject,
    keys: keys.length,
    recalls,
    mean: recalls.reduce((a, b) => a + b, 0) / (recalls.length || 1),
    costUsd: runs.reduce((sum, r) => sum + r.review.costUsd + r.grade.costUsd, 0),
  };
}
```

```js
#!/usr/bin/env node
// Runs /comb:review on a benchmark subject N times and grades every run against the subject's answer key.
// Usage: node bench/run.mjs <subject> [--runs N]
// Needs `npm run build` and the `claude` CLI. Each run costs roughly $3–5 (one review and one grader).
import { spawnSync } from 'node:child_process';
import { chmodSync, cpSync, mkdirSync, mkdtempSync, readFileSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { dirname, join, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { answerKeys, graderPrompt, parseGrade, resultOf, summarize } from './lib.mjs';

const repo = resolve(dirname(fileURLToPath(import.meta.url)), '..');
const [subject, ...rest] = process.argv.slice(2);
const at = rest.indexOf('--runs');
const runs = at >= 0 ? Number(rest[at + 1]) : 1;
if (!subject || !(runs > 0)) {
  console.error('usage: node bench/run.mjs <subject> [--runs N]');
  process.exit(2);
}
const dir = join(repo, 'bench', subject);
const config = JSON.parse(readFileSync(join(dir, 'bench.json'), 'utf8'));
const answerKey = readFileSync(join(dir, 'expected-highs.md'), 'utf8');
const keys = answerKeys(answerKey);
const spec = readFileSync(join(dir, config.spec), 'utf8');

const tmp = mkdtempSync(join(tmpdir(), `comb-bench-${subject}-`));
mkdirSync(join(tmp, 'bin'));
writeFileSync(join(tmp, 'bin', 'comb'), `#!/bin/sh\nexec node ${join(repo, 'dist', 'cli', 'index.js')} "$@"\n`);
chmodSync(join(tmp, 'bin', 'comb'), 0o755);
const env = { ...process.env, PATH: `${join(tmp, 'bin')}:${process.env.PATH}` };

const sh = (cmd, cwd) => {
  const r = spawnSync('sh', ['-c', cmd], { cwd, env, encoding: 'utf8' });
  if (r.status !== 0) throw new Error(`${cmd} failed: ${r.stderr}`);
};
const claude = (prompt, cwd, tools) => {
  const r = spawnSync('claude', ['-p', prompt, '--permission-mode', 'acceptEdits', '--allowedTools', ...tools, '--output-format', 'stream-json', '--verbose'], {
    cwd,
    env,
    encoding: 'utf8',
    maxBuffer: 1 << 28,
  });
  const result = resultOf(r.stdout ?? '');
  if (!result) throw new Error(`claude produced no result event: ${r.stderr}`);
  return result;
};

const results = [];
for (let i = 1; i <= runs; i++) {
  const work = join(tmp, `run-${i}`);
  cpSync(join(dir, 'project'), work, { recursive: true });
  sh('git init -q && comb init --language en >/dev/null && git add -A && git commit -qm base', work);
  const review = claude(`/comb:review ${config.change}`, work, ['Bash(comb:*)', 'Read', 'Write', 'Edit', 'Glob', 'Grep']);
  const questions = readFileSync(join(work, 'comb', 'changes', config.change, 'questions.md'), 'utf8');
  const grade = claude(graderPrompt({ spec, answerKey, questions }), work, ['Read']);
  const matches = parseGrade(grade.text, keys);
  results.push({ run: i, workDir: work, review: { ...review, text: undefined }, grade: { costUsd: grade.costUsd, minutes: grade.minutes }, matches });
  console.log(`run ${i}: ${matches.filter((m) => m.question).length}/${keys.length}  (review $${review.costUsd.toFixed(2)}, ${review.minutes} min, ${review.toolErrors} tool errors)`);
}
const summary = summarize(subject, results, keys);
mkdirSync(join(repo, 'bench', 'results'), { recursive: true });
const out = join(repo, 'bench', 'results', `${subject}-${new Date().toISOString().replace(/[:.]/g, '-')}.json`);
writeFileSync(out, `${JSON.stringify({ ...summary, pass: config.pass, runs: results }, null, 2)}\n`);
console.log(`mean ${summary.mean.toFixed(1)}/${keys.length} (pass bar ${config.pass}), $${summary.costUsd.toFixed(2)}; written to ${out}`);
```
